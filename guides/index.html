第三步：创建你的第一个智能体 ​

python
# 简单的智能体示例
from ai_agent import Agent

# 创建智能体实例
agent = Agent(
    name="我的助手",
    model="gpt-4",
    system_prompt="你是一个友好的AI助手"
)

# 与智能体对话
response = agent.chat("你好，请介绍一下自己")
print(response)
